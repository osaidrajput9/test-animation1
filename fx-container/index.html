const fx = new FXContainer(document.querySelector('.fx'), {
  effect: 'ring',        // 'ring' | 'coverflow' | 'deck' | 'helix' | any registered effect
  duration: 0.9,         // step animation length (s)
  ease: 'power3.out',
  tilt: true,            // pointer-driven 3D tilt of the whole scene
  autoplay: 0,           // seconds between steps, 0 = off
  accent: '#3D5AFE',
  onChange: (index) => console.log('active', index)
});

fx.next(); fx.prev(); fx.show(3);     // navigation
fx.use('coverflow');                  // morph to another layout
fx.setColor('#FF6B4A');               // animated re-theme
fx.flip();                            // flip the active card
fx.setAutoplay(4); fx.setTilt(false);
fx.destroy();                         // remove listeners, clear inline styles

// Add your own layout: return a pose for each item
FXContainer.register('wave', ({ d, w }) => ({
  x: d * w * 0.8, y: Math.sin(d) * 60, rotationZ: d * 6,
  opacity: 1 - Math.abs(d) * 0.25
}));
